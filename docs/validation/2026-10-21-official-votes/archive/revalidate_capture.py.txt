"""Offline revalidation of a finished real House HTTP capture; never fetches.
Original receipts/timestamps/bodies and failed manifests remain immutable.
"""
import argparse, copy, hashlib, json, re, shutil, sys
from collections import Counter
from datetime import datetime, timezone
from pathlib import Path
ROOT=Path(__file__).resolve().parents[3]
sys.path.insert(0,str(ROOT/'tools/congress-collector'))
import adapter

def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def require(condition,message):
 if not condition:raise ValueError(message)
def revalidate_record(record,acquisition_root,validator):
 r=copy.deepcopy(record);p=r['response'];year=r['year'];roll=r['roll'];congress=r['congress'];session=r['session']
 require(r['chamber']=='house' and session in (1,2) and year==1787+2*congress+session-1,'invalid House identity')
 require(r['key']==f'{congress}-{year}/h{roll}','roll key differs')
 adapter.iso_date(r['date']);require(r['date'].startswith(str(year)+'-'),'date year differs')
 url=f'https://clerk.house.gov/evs/{year}/roll{roll:03d}.xml'
 require(r['source_url']==url and p['url']==url and p['final_url']==url,'response URL differs from canonical source')
 require(r['in_window'] is True and r['attempted'] is True and bool(r['attempted_at']),'response was not attempted')
 require(p['http_status']==200 and p['body_complete'] is True and p['read_error'] is None,'HTTP failure or incomplete response')
 require(bool(p['attempted_at']) and bool(p['received_at']),'capture timestamps missing')
 name=p['path'];require(isinstance(name,str) and re.fullmatch(r'responses/[0-9]{4}-h[0-9]+\.body',name),'unsafe response path')
 root=Path(acquisition_root).resolve();path=root/name
 require(not path.is_symlink() and not path.parent.is_symlink() and path.resolve().is_relative_to(root),'unsafe source body')
 b=path.read_bytes();require(0<len(b)<=adapter.MAX_RESPONSE_BYTES and len(b)==p['bytes'] and hashlib.sha256(b).hexdigest()==p['sha256'],'retained body hash/size differs')
 r['original_outcome']=r['outcome'];r['original_reason']=r['reason'];r['original_metadata']=copy.deepcopy(r['metadata'])
 r['metadata']=validator(b,r);r['metadata_changed']=r['metadata']!=r['original_metadata'];r['outcome']='acquired';r['reason']=None
 return r

def revalidate(source,dest):
 interfaces={l.split(':')[0].strip() for l in Path('/proc/self/net/dev').read_text().splitlines() if ':' in l}
 require(interfaces<={'lo'},'run revalidation inside unshare -rn')
 source=Path(source).resolve();dest=Path(dest).absolute()
 require(not dest.exists() and not dest.is_symlink(),'destination already exists')
 require(not any(p.is_symlink() for p in dest.parents),'symlink destination parent')
 require(dest.is_relative_to(ROOT/'artifacts') and not dest.is_relative_to(source),'destination must be a new candidate artifact directory outside source')
 for p in source.rglob('*'):require(not p.is_symlink(),'source capture contains symlink')
 run=json.loads((source/'window-run.json').read_text());m=json.loads((source/'acquisition/manifest.json').read_text())
 require(run['chamber']=='house' and run['mode']=='window' and run['status'] in ['complete','incomplete'] and bool(run['finished_at']),'source is not a finished House window capture')
 require(run['manifest_path']=='acquisition/manifest.json' and sha(source/run['manifest_path'])==run['manifest_sha256'],'original manifest hash differs')
 require(run['discovery']['inventory_path']=='discovery/inventory.json','unexpected inventory path')
 inventory=adapter.validate_inventory(source/run['discovery']['inventory_path'],run['discovery']['inventory_sha256'])
 require(list(inventory.window)==run['window'] and run['months']==48 and run['window']==[adapter.subtract_months(run['as_of'],48),run['as_of']],'not an exact 48-month capture')
 require(m['status'] in ['acquisition_complete','incomplete'] and bool(m['finished_at']),'manifest not finished')
 require(m['required_rolls']==len(inventory.selected)>0,'inventory/manifest required count differs')
 require(len(m['records'])==len(inventory.records),'inventory/manifest total differs')
 require(len({r['key'] for r in m['records']})==len(m['records']),'duplicate captured record')
 expected={r['key']:r for r in inventory.records}
 for r in m['records']:
  e=expected.get(r['key']);require(e is not None and all(r[k]==e[k] for k in ['congress','session','year','roll','date']),'captured identity differs from inventory')
  require(r['in_window']==(run['window'][0]<=r['date']<=run['window'][1]),'wrong in-window selection')
  if r['in_window']:require(r['attempted'] is True and r['response'] is not None,'required roll was never captured')
  else:require(r['outcome']=='excluded_outside_window' and r['attempted'] is False and r['response'] is None,'invalid outside-window exclusion')
 before={'window_run_sha256':sha(source/'window-run.json'),'manifest_sha256':sha(source/'acquisition/manifest.json')}
 started=datetime.now(timezone.utc).isoformat()
 shutil.copytree(source,dest,copy_function=shutil.copy2)
 revised=copy.deepcopy(m);failures=[]
 for i,r in enumerate(m['records']):
  if not r['in_window']:continue
  try:revised['records'][i]=revalidate_record(r,source/'acquisition',adapter.validate_xml)
  except (ValueError,OSError,KeyError,TypeError,adapter.ET.ParseError) as e:
   revised['records'][i].update(original_outcome=r['outcome'],original_reason=r['reason'],original_metadata=copy.deepcopy(r['metadata']))
   revised['records'][i].update(outcome='malformed_source',reason=f'offline revalidation failed: {e}',metadata=None)
   failures.append({'key':r['key'],'reason':str(e)})
 revised['counts']=dict(Counter(r['outcome'] for r in revised['records']));revised['status']='acquisition_complete' if not failures else 'incomplete'
 revised['normalized_votes_validated']=False;revised['member_coverage_validated']=False;revised['bill_resolution_validated']=False
 validated_at=datetime.now(timezone.utc).isoformat()
 provenance={'operation':'offline_revalidation_of_real_http_capture','original_run':str(source),**before}
 for document,original in [(revised,m)]:
  document.update(derived_from=provenance,revalidated_at=validated_at,source_capture_started_at=original['started_at'],source_capture_finished_at=original['finished_at'],started_at=started,finished_at=validated_at,processing_semantics='offline_revalidation_of_previously_downloaded_bodies')
 (dest/'acquisition/manifest.json').write_text(json.dumps(revised,sort_keys=True,indent=2)+'\n')
 derived=copy.deepcopy(run);derived['manifest_sha256']=sha(dest/'acquisition/manifest.json');derived['status']='complete' if not failures else 'incomplete'
 derived.update(derived_from=provenance,revalidated_at=validated_at,source_capture_started_at=run['started_at'],source_capture_finished_at=run['finished_at'],started_at=started,finished_at=validated_at)
 derived['reason']='Offline revalidation of retained real HTTP capture; no HTTP request or synthetic receipt. See revalidation.json.'
 (dest/'window-run.json').write_text(json.dumps(derived,sort_keys=True,indent=2)+'\n')
 require(before['window_run_sha256']==sha(source/'window-run.json') and before['manifest_sha256']==sha(source/'acquisition/manifest.json'),'original evidence changed')
 proof={'operation':'offline_revalidation_of_real_http_capture','validated_at':validated_at,'original_run':str(source),'original_hashes':before,'original_candidate':m.get('collector_revision'),'original_candidate_evidence':'source manifest collector_revision; null means unrecorded, not inferred','new_http_requests':0,'original_receipts_and_timestamps_preserved':True,'revalidation_script_path':str(Path(__file__).resolve()),'revalidation_script_sha256':sha(__file__),'validator_sha256':{str(p.relative_to(ROOT)):sha(p) for p in (ROOT/'tools/congress-collector').glob('*.py')},'required_rolls':m['required_rolls'],'failures':failures,'derived_manifest_sha256':derived['manifest_sha256'],'derived_window_run_sha256':sha(dest/'window-run.json'),'original_outcome_counts':dict(Counter(r['outcome'] for r in m['records'])),'metadata_changed_previously_acquired':[r['key'] for r in revised['records'] if r.get('original_outcome')=='acquired' and r.get('metadata_changed')]}
 (dest/'revalidation.json').write_text(json.dumps(proof,indent=2)+'\n')
 return {'status':derived['status'],'run_dir':str(dest),'required_rolls':m['required_rolls'],'failures':failures}

if __name__=='__main__':
 p=argparse.ArgumentParser(description=__doc__);p.add_argument('--source',required=True);p.add_argument('--output',required=True);a=p.parse_args()
 try:
  result=revalidate(a.source,a.output);print(json.dumps(result));sys.exit(0 if result['status']=='complete' else 1)
 except Exception as e:print(json.dumps({'status':'failed','reason':str(e)}),file=sys.stderr);sys.exit(1)
